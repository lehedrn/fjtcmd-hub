## 2. 表结构 (DDL)

### 2.1 主表结构

```sql
CREATE TABLE cms_article (
  id              BIGINT          NOT NULL      AUTO_INCREMENT COMMENT '文章 ID',
  title           VARCHAR(200)    NOT NULL      COMMENT '文章标题',
  author          VARCHAR(50)     NOT NULL      COMMENT '作者',
  content         LONGTEXT        NOT NULL      COMMENT '文章内容',
  status          CHAR(1)         DEFAULT '0'   COMMENT '状态（0 草稿 1 待审核 2 已发布 3 已下架）',
  dept_id         BIGINT          DEFAULT NULL  COMMENT '部门 ID（数据权限）',
  publish_time    DATETIME        DEFAULT NULL  COMMENT '发布时间',
  publish_by      VARCHAR(64)     DEFAULT NULL  COMMENT '发布人',
  audit_time      DATETIME        DEFAULT NULL  COMMENT '审核时间',
  audit_by        VARCHAR(64)     DEFAULT NULL  COMMENT '审核人',
  audit_remark    VARCHAR(500)    DEFAULT NULL  COMMENT '审核意见',
  create_time     DATETIME        DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  update_time     DATETIME        DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  create_by       VARCHAR(64)     DEFAULT ''    COMMENT '创建人',
  update_by       VARCHAR(64)     DEFAULT ''    COMMENT '更新人',
  remark          VARCHAR(500)    DEFAULT ''    COMMENT '备注',
  del_flag        CHAR(1)         DEFAULT '0'   COMMENT '删除标志',
  PRIMARY KEY (id),
  KEY idx_status (status),
  KEY idx_dept_id (dept_id),
  KEY idx_publish_time (publish_time)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COMMENT='文章信息表';
```

## 3. 业务规则

### 3.1 状态流转规则

```
草稿 (0) → 待审核 (1) → 已发布 (2) → 已下架 (3)
              ↓
         驳回 → 草稿 (0)
```

**状态说明**：

| 状态 | 编码 | 说明 | 可执行操作 |
|------|------|------|-----------|
| 草稿 | 0 | 文章初始状态，可编辑 | 修改、提交审核、删除 |
| 待审核 | 1 | 等待审核员审核 | 审核通过、驳回 |
| 已发布 | 2 | 文章已发布，可访问 | 下架 |
| 已下架 | 3 | 文章已下架，不可访问 | 重新发布、删除 |

### 3.2 业务校验规则

| 规则 | 说明 | 触发时机 |
|------|------|---------|
| 标题唯一性 | 同一分类下标题不能重复 | 新增/修改 |
| 内容非空 | 文章内容不能为空 | 提交审核 |
| 状态校验 | 只有草稿状态可提交审核 | 提交审核 |
| 权限校验 | 编辑只能修改自己的文章 | 修改 |
| 审核权限 | 只有审核员可审核 | 审核 |
| 发布权限 | 只有管理员可发布 | 发布 |

### 3.3 数据权限规则

**部门数据隔离**：
- 编辑：只能看到本部门的文章
- 审核员：可以审核本部门的文章
- 管理员：可以看到所有部门的文章

**实现方式**：
```java
@DataScope(deptAlias = "d", userAlias = "u")
public List<Article> selectArticleList(Article article) {
    return articleMapper.selectArticleList(article);
}
```

### 4.2 接口详细设计

#### 4.2.1 提交审核

**请求**：
```http
PUT /cms/article/audit/1
Authorization: Bearer {token}
```

**响应**：
```json
{
  "code": 200,
  "msg": "提交审核成功"
}
```

**业务逻辑**：
```java
@Transactional
public int submitAudit(Long articleId) {
    Article article = articleMapper.selectArticleById(articleId);
    
    // 1. 状态校验
    if (!"0".equals(article.getStatus())) {
        throw new ServiceException("只有草稿状态可以提交审核");
    }
    
    // 2. 内容校验
    if (StringUtils.isEmpty(article.getContent())) {
        throw new ServiceException("文章内容不能为空");
    }
    
    // 3. 更新状态
    article.setStatus("1");
    article.setUpdateBy(SecurityUtils.getUsername());
    article.setUpdateTime(DateUtils.getNowDate());
    
    return articleMapper.updateArticle(article);
}
```

#### 4.2.2 审核通过

**请求**：
```http
PUT /cms/article/publish/1
Authorization: Bearer {token}
```

**响应**：
```json
{
  "code": 200,
  "msg": "审核通过，文章已发布"
}
```

**业务逻辑**：
```java
@Transactional
public int publishArticle(Long articleId) {
    Article article = articleMapper.selectArticleById(articleId);
    
    // 1. 权限校验
    if (!SecurityUtils.isAdmin(SecurityUtils.getUserId())) {
        throw new ServiceException("没有发布权限");
    }
    
    // 2. 状态校验
    if (!"1".equals(article.getStatus())) {
        throw new ServiceException("只有待审核状态可以发布");
    }
    
    // 3. 更新状态
    article.setStatus("2");
    article.setPublishTime(DateUtils.getNowDate());
    article.setPublishBy(SecurityUtils.getUsername());
    article.setAuditTime(DateUtils.getNowDate());
    article.setAuditBy(SecurityUtils.getUsername());
    article.setAuditRemark("审核通过");
    
    return articleMapper.updateArticle(article);
}
```

**请求**：

public int rejectArticle(Long articleId, String auditRemark) {

    // 1. 状态校验
    if (!"1".equals(article.getStatus())) {
        throw new ServiceException("只有待审核状态可以驳回");
    }
    
    // 2. 更新状态
    article.setStatus("0");
    article.setAuditTime(DateUtils.getNowDate());
    article.setAuditBy(SecurityUtils.getUsername());
    article.setAuditRemark(auditRemark);
    
    return articleMapper.updateArticle(article);
}

## 6. 实现要点

### 6.1 Service 层实现

```java
@Service
public class ArticleServiceImpl implements IArticleService {
    
    @Autowired
    private ArticleMapper articleMapper;
    
    /**
     * 查询文章列表（带数据权限）
     */
    @Override
    @DataScope(deptAlias = "d", userAlias = "u")
    public List<Article> selectArticleList(Article article) {
        return articleMapper.selectArticleList(article);
    }
    
    /**
     * 新增文章
     */
    @Override
    public int insertArticle(Article article) {
        // 设置初始状态为草稿
        article.setStatus("0");
        article.setCreateBy(SecurityUtils.getUsername());
        article.setCreateTime(DateUtils.getNowDate());
        return articleMapper.insertArticle(article);
    }
    
    /**
     * 修改文章
     */
    @Override
    public int updateArticle(Article article) {
        // 校验：只有草稿状态可以修改
        Article existing = articleMapper.selectArticleById(article.getId());
        if (!"0".equals(existing.getStatus())) {
            throw new ServiceException("只有草稿状态的文章可以修改");
        }
        
        // 校验：只能修改自己的文章
        if (!existing.getCreateBy().equals(SecurityUtils.getUsername())) {
            throw new ServiceException("只能修改自己创建的文章");
        }
        
        article.setUpdateBy(SecurityUtils.getUsername());
        article.setUpdateTime(DateUtils.getNowDate());
        return articleMapper.updateArticle(article);
    }
}
```

### 6.2 Controller 层实现

```java
@RestController
@RequestMapping("/cms/article")
public class ArticleController extends BaseController {
    
    @Autowired
    private IArticleService articleService;
    
    /**
     * 提交审核
     */
    @PreAuthorize("@ss.hasPermi('cms:article:audit')")
    @Log(title = "文章管理", businessType = BusinessType.UPDATE)
    @PutMapping("/audit/{id}")
    public AjaxResult submitAudit(@PathVariable Long id) {
        return toAjax(articleService.submitAudit(id));
    }
    
    /**
     * 审核通过
     */
    @PreAuthorize("@ss.hasPermi('cms:article:publish')")
    @Log(title = "文章管理", businessType = BusinessType.UPDATE)
    @PutMapping("/publish/{id}")
    public AjaxResult publish(@PathVariable Long id) {
        return toAjax(articleService.publishArticle(id));
    }
    
    /**
     * 审核驳回
     */
    @PreAuthorize("@ss.hasPermi('cms:article:audit')")
    @Log(title = "文章管理", businessType = BusinessType.UPDATE)
    @PutMapping("/reject/{id}")
    public AjaxResult reject(@PathVariable Long id, @RequestBody Article article) {
        return toAjax(articleService.rejectArticle(id, article.getAuditRemark()));
    }
    
    /**
     * 下架文章
     */
    @PreAuthorize("@ss.hasPermi('cms:article:offline')")
    @Log(title = "文章管理", businessType = BusinessType.UPDATE)
    @PutMapping("/offline/{id}")
    public AjaxResult offline(@PathVariable Long id) {
        return toAjax(articleService.offlineArticle(id));
    }
}
```

## 8. 注意事项

1. **状态流转**：必须严格按照状态机流转，不允许跳级
2. **权限控制**：不同角色的操作权限必须严格控制
3. **数据权限**：使用 @DataScope 注解实现部门数据隔离
4. **事务控制**：状态变更必须使用事务，确保数据一致性
5. **操作日志**：关键操作必须记录操作日志
6. **异常处理**：业务校验失败必须抛出 ServiceException
