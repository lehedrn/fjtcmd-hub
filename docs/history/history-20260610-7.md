```yaml

  sys_goods:
    functionName: 商品信息
    columns:
      # 客户ID - 不展示在搜索和列表，但插入时自动带上
      customer_id:
        isList: false         # 列表不显示
        isQuery: false        # 搜索表单不显示

        queryType: LIKE
```

```
    ↓
```
